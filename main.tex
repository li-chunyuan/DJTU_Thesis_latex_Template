% ======================%一些说明：=============================
% 这是一个通用的中文文章 LaTeX 模板，适用于：

% - 课程报告、项目文档
% - 学术论文、科技文章
% 请确保 config.tex 中的相关设置与 main.tex 一致，避免冲突报错。



% ==================================================================================
% ① 文档类：ctexart 自带中文支持；A4 纸，12pt 字号
\documentclass[a4paper,12pt]{ctexart}%ctexart 是专为中文排版准备的文档类 ,12pt 对应小四字号

\input{latex/config}    % 导入配置文件config.tex

% ==================================单独调试区域=====================================

%\includeonly{latex/chapter3/basic}      %单独调试某一章节

% ==================================================================================

\begin{document}

% ====================================封面===========================================

% \maketitle                这条命令是使用 LaTeX 默认的封面命令（已注释）
\input{latex/cover/cover}   % 现在改用自定义封面页内容
\newpage%立即结束当前页，从下一页开始排版

% ====================================目录===========================================

\pagestyle{empty}   % 先关闭页眉页脚，属于永久开关（直到下次打开）
\tableofcontents    % 生成目录（此时目录页就没有页眉页脚了）

% \thispagestyle{empty} % 如果用别的文档类，可能需要加上这一行，确保目录页没有页眉页脚
\newpage             % 目录结束，换页

% ====================================正文===========================================

% 2. 正文部分 (要求：有页眉、页码从正文开始计数)
\pagestyle{reportstyle} % 应用上面定义的样式
\renewcommand{\sectionmark}[1]{\markboth{#1}{}} % 仅使用标题作为页眉标记
\setcounter{page}{1}    % 强制将页码重置为 1

% ---------------------正文内容 ------------------------------------
\include{latex/chapter1/1}      % 第一章 引言
\include{latex/chapter2/2}      % 第二章 相关理论与技术
\include{latex/chapter3/3}      % 第三章 图表示例
\include{latex/chapter4/4}      % 第四章 代码示例
\include{latex/chapter5/5}      % 第五章 分析与讨论
\include{latex/chapter6/6}      % 第六章 总结与展望




\end{document}
